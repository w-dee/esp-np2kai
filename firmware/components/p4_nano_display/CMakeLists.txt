idf_component_register(
    SRCS "p4_nano_display.cpp" "p4_nano_display_pattern.cpp"
         "p4_nano_display_transform.cpp" "p4_nano_display_transform_pattern.cpp"
         "p4_nano_jd9365_safe.c"
    INCLUDE_DIRS "include" "include/p4_nano_display"
    REQUIRES p4_nano_board esp_lcd esp_hw_support esp_mm esp_driver_gpio
             esp_driver_i2c esp_psram heap freertos esp_timer esp_lcd_jd9365_10_1)

if(P4_NANO_EXACT2X_SCALER_BENCHMARK_PROFILE OR
   "$ENV{P4_NANO_EXACT2X_SCALER_BENCHMARK_PROFILE}" STREQUAL "1" OR
   P4_NANO_EXACT2X_INTERNAL_SOURCE_BENCHMARK_PROFILE OR
   "$ENV{P4_NANO_EXACT2X_INTERNAL_SOURCE_BENCHMARK_PROFILE}" STREQUAL "1" OR
   P4_NANO_EXACT2X_GROUPED_STORE_BENCHMARK_PROFILE OR
   "$ENV{P4_NANO_EXACT2X_GROUPED_STORE_BENCHMARK_PROFILE}" STREQUAL "1" OR
   P4_NANO_PPA_PIE_OVERLAP_BENCHMARK_PROFILE OR
   "$ENV{P4_NANO_PPA_PIE_OVERLAP_BENCHMARK_PROFILE}" STREQUAL "1" OR
   P4_NANO_PPA_PIE_BURST_BENCHMARK_PROFILE OR
   "$ENV{P4_NANO_PPA_PIE_BURST_BENCHMARK_PROFILE}" STREQUAL "1" OR
   P4_NANO_EXACT2X_DMA2D_CORRECTNESS_PROFILE OR
   "$ENV{P4_NANO_EXACT2X_DMA2D_CORRECTNESS_PROFILE}" STREQUAL "1" OR
   P4_NANO_EXACT2X_DMA2D_BENCHMARK_PROFILE OR
   "$ENV{P4_NANO_EXACT2X_DMA2D_BENCHMARK_PROFILE}" STREQUAL "1")
    target_sources(${COMPONENT_LIB} PRIVATE
        p4_nano_display_exact2x.cpp p4_nano_display_exact2x_pie.S)
endif()

set(P4_NANO_REFRESH_VISUAL_PROFILE_SELECTED "")
if(P4_NANO_REFRESH_VISUAL_PROFILE)
    set(P4_NANO_REFRESH_VISUAL_PROFILE_SELECTED
        "${P4_NANO_REFRESH_VISUAL_PROFILE}")
elseif(DEFINED ENV{P4_NANO_REFRESH_VISUAL_PROFILE})
    set(P4_NANO_REFRESH_VISUAL_PROFILE_SELECTED
        "$ENV{P4_NANO_REFRESH_VISUAL_PROFILE}")
endif()
if(NOT P4_NANO_REFRESH_VISUAL_PROFILE_SELECTED STREQUAL "")
    target_sources(${COMPONENT_LIB} PRIVATE p4_nano_refresh_visual.cpp)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_REFRESH_VISUAL_PROFILE=1)
    if(P4_NANO_REFRESH_VISUAL_PROFILE_SELECTED STREQUAL "baseline")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_REFRESH_VISUAL_BASELINE_PROFILE=1)
    elseif(P4_NANO_REFRESH_VISUAL_PROFILE_SELECTED STREQUAL "lower1")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_REFRESH_VISUAL_LOWER1_PROFILE=1)
    elseif(P4_NANO_REFRESH_VISUAL_PROFILE_SELECTED STREQUAL "lower2")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_REFRESH_VISUAL_LOWER2_PROFILE=1)
    else()
        message(FATAL_ERROR
            "P4_NANO_REFRESH_VISUAL_PROFILE must be baseline, lower1, or lower2")
    endif()
endif()

set(P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE_SELECTED "")
if(P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE)
    set(P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE_SELECTED
        "${P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE}")
endif()
if(NOT P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE_SELECTED STREQUAL "")
    if(NOT P4_NANO_EXACT2X_INTERNAL_SOURCE_BENCHMARK_PROFILE AND
       NOT "$ENV{P4_NANO_EXACT2X_INTERNAL_SOURCE_BENCHMARK_PROFILE}" STREQUAL "1" AND
       NOT P4_NANO_EXACT2X_GROUPED_STORE_BENCHMARK_PROFILE AND
       NOT "$ENV{P4_NANO_EXACT2X_GROUPED_STORE_BENCHMARK_PROFILE}" STREQUAL "1" AND
       NOT P4_NANO_EXACT2X_DMA2D_BENCHMARK_PROFILE AND
       NOT "$ENV{P4_NANO_EXACT2X_DMA2D_BENCHMARK_PROFILE}" STREQUAL "1" AND
       NOT P4_NANO_PPA_PIE_OVERLAP_BENCHMARK_PROFILE AND
       NOT "$ENV{P4_NANO_PPA_PIE_OVERLAP_BENCHMARK_PROFILE}" STREQUAL "1" AND
       NOT P4_NANO_PPA_PIE_BURST_BENCHMARK_PROFILE AND
       NOT "$ENV{P4_NANO_PPA_PIE_BURST_BENCHMARK_PROFILE}" STREQUAL "1")
        message(FATAL_ERROR
            "P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE requires the P10F internal-source benchmark")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE=1)
    if(P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE_SELECTED STREQUAL "baseline")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_BENCHMARK_DISPLAY_REFRESH_BASELINE_PROFILE=1)
    elseif(P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE_SELECTED STREQUAL "lower2")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_BENCHMARK_DISPLAY_REFRESH_LOWER2_PROFILE=1)
    else()
        message(FATAL_ERROR
            "P4_NANO_BENCHMARK_DISPLAY_REFRESH_PROFILE must be baseline or lower2")
    endif()
endif()

target_compile_options(${COMPONENT_LIB} PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-std=gnu++20>"
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>"
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>"
    -Wall -Wextra)

if(P4_NANO_RUNTIME_EMU_BACKEND)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_RUNTIME_EMU_BACKEND=1)
endif()

set(P4_NANO_DISPLAY_TRANSFORM_OPT "debug" CACHE STRING
    "Optimization mode for the P4-NANO transform translation unit")
set_property(CACHE P4_NANO_DISPLAY_TRANSFORM_OPT PROPERTY STRINGS debug o2)
if(P4_NANO_DISPLAY_TRANSFORM_OPT STREQUAL "o2")
    set_source_files_properties(
        p4_nano_display_transform.cpp p4_nano_display_exact2x.cpp
        PROPERTIES COMPILE_OPTIONS "-O2")
elseif(NOT P4_NANO_DISPLAY_TRANSFORM_OPT STREQUAL "debug")
    message(FATAL_ERROR
        "P4_NANO_DISPLAY_TRANSFORM_OPT must be debug or o2, got: "
        "${P4_NANO_DISPLAY_TRANSFORM_OPT}")
endif()

        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_PSRAM_BANDWIDTH_OPERATION_READ=1)
    elseif(P4_NANO_PSRAM_BANDWIDTH_OPERATION STREQUAL "write16")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_PSRAM_BANDWIDTH_OPERATION_WRITE16=1)
    elseif(P4_NANO_PSRAM_BANDWIDTH_OPERATION STREQUAL "write32")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_PSRAM_BANDWIDTH_OPERATION_WRITE32=1)
    elseif(P4_NANO_PSRAM_BANDWIDTH_OPERATION STREQUAL "memcpy")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_PSRAM_BANDWIDTH_OPERATION_MEMCPY=1)
    elseif(P4_NANO_PSRAM_BANDWIDTH_OPERATION STREQUAL "row-copy")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_PSRAM_BANDWIDTH_OPERATION_ROW_COPY=1)
    elseif(P4_NANO_PSRAM_BANDWIDTH_OPERATION STREQUAL "proxy")
        target_compile_definitions(${COMPONENT_LIB} PRIVATE
            P4_NANO_PSRAM_BANDWIDTH_OPERATION_PROXY=1)
    else()
        message(FATAL_ERROR
            "P4_NANO_PSRAM_BANDWIDTH_OPERATION must be read, write16, write32, memcpy, row-copy, or proxy")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_PSRAM_BANDWIDTH_BENCHMARK_PROFILE=1)
endif()

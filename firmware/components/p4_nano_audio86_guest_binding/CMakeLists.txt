cmake_minimum_required(VERSION 3.16)

get_filename_component(P4_NANO_AUDIO86_BINDING_ROOT
                       "${CMAKE_CURRENT_LIST_DIR}/../../.." REALPATH)
set(P4_NANO_AUDIO86_VENDOR_SRC
    "${P4_NANO_AUDIO86_BINDING_ROOT}/third_party/np2kai/src")

set(P4_NANO_AUDIO86_BINDING_REQUIRES
    np2core np2host np2runtime np2audio86_fixture np2audio86_guest_adapter np2opngen_fixture
    np2pcm_output p4_nano_audio86_notifications p4_nano_audio86_live_service
    p4_nano_audio86_physical_sink freertos esp_system esp_timer heap)

idf_component_register(
    SRCS "p4_nano_audio86_guest_binding.cpp"
    INCLUDE_DIRS "include" "${P4_NANO_AUDIO86_VENDOR_SRC}"
                 "${P4_NANO_AUDIO86_VENDOR_SRC}/i286c"
                 "${P4_NANO_AUDIO86_BINDING_ROOT}/host/compat"
    REQUIRES ${P4_NANO_AUDIO86_BINDING_REQUIRES})

target_compile_options(${COMPONENT_LIB} PRIVATE
    -std=gnu++20 -fno-exceptions -fno-rtti -Wall -Wextra -Werror
    "SHELL:-iquote ${P4_NANO_AUDIO86_VENDOR_SRC}"
    "SHELL:-iquote ${P4_NANO_AUDIO86_VENDOR_SRC}/cbus"
    "SHELL:-iquote ${P4_NANO_AUDIO86_VENDOR_SRC}/io")

target_compile_definitions(${COMPONENT_LIB} PRIVATE
    DISABLE_SOUND BYTESEX_LITTLE NP2_ASYNC_AUDIO86=1)
if(P4_NANO_AUDIO86_PRESSURE_SCENARIO)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_PRESSURE_SCENARIO=${P4_NANO_AUDIO86_PRESSURE_SCENARIO})
endif()
if(P4_NANO_AUDIO86_FAILURE_KIND)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_FAILURE_KIND=${P4_NANO_AUDIO86_FAILURE_KIND})
endif()
if(P4_NANO_AUDIO86_PCM_OUTPUT_PROFILE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_PCM_OUTPUT_PROFILE=1)
endif()
if(P4_NANO_AUDIO86_SUSTAINED_PROFILE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_SUSTAINED_PROFILE=1)
endif()
if(P4_NANO_AUDIO86_TERMINAL_POST_PCM_FAILURE_TEST OR
   "$ENV{P4_NANO_AUDIO86_TERMINAL_POST_PCM_FAILURE_TEST}" STREQUAL "1")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_TERMINAL_POST_PCM_FAILURE_TEST=1)
endif()
if(P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST)
    set(P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST_VALUE
        ${P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST})
elseif(DEFINED ENV{P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST})
    set(P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST_VALUE
        $ENV{P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST})
endif()
if(P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST_VALUE)
    if(NOT P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST_VALUE MATCHES "^[12]$")
        message(FATAL_ERROR
            "terminal publication test must be 1 (success) or 2 (partial failure)")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST=${P4_NANO_AUDIO86_TERMINAL_PUBLICATION_TEST_VALUE})
endif()
if(P4_NANO_AUDIO86_SUSTAINED_PHYSICAL_PROFILE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_SUSTAINED_PHYSICAL_PROFILE=1)
endif()
if(P4_NANO_AUDIO86_PHYSICAL_I2S_PROFILE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_PHYSICAL_I2S_PROFILE=1)
endif()
if(P4_NANO_AUDIO86_PHYSICAL_I2S_PROFILE AND
   NOT P4_NANO_AUDIO86_PHYSICAL_LIFECYCLE_TEST_PROFILE)
    if(NOT P4_AUDIO86_GIT_SHA)
        message(FATAL_ERROR
            "physical Audio 86 evidence requires P4_AUDIO86_GIT_SHA")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_AUDIO86_GIT_SHA="${P4_AUDIO86_GIT_SHA}")
endif()
if(P4_NANO_AUDIO86_PHYSICAL_SHORT_PROFILE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_PHYSICAL_SHORT_PROFILE=1)
endif()
if(P4_NANO_AUDIO86_PHYSICAL_LIFECYCLE_TEST_PROFILE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_PHYSICAL_I2S_PROFILE=1
        P4_NANO_AUDIO86_PHYSICAL_LIFECYCLE_TEST_PROFILE=${P4_NANO_AUDIO86_PHYSICAL_LIFECYCLE_TEST_PROFILE})
endif()
if(P4_NANO_AUDIO86_PCM_PARTIAL_EOS_PROFILE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_PCM_PARTIAL_EOS_PROFILE=1)
endif()
if(P4_NANO_AUDIO86_PCM_LIFECYCLE_SCENARIO)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        P4_NANO_AUDIO86_PCM_LIFECYCLE_SCENARIO=${P4_NANO_AUDIO86_PCM_LIFECYCLE_SCENARIO})
endif()
